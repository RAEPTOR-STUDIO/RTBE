cmake_minimum_required(VERSION 3.16)
project(RTBE LANGUAGES C CXX)
find_package(RaeptorCogs CONFIG REQUIRED)
find_package(Cog CONFIG REQUIRED)
CogCreate(${PROJECT_NAME}
  src/RaeptorCogs/${PROJECT_NAME}/rtbe.c
)
CogVersion(${PROJECT_NAME} 1.0.0)
CogLoad("../../ExampleCog" ExampleCog)
CogLoad("../../PrimeMover" PrimeMover)
CogLoad("../../RaylibCog" RaylibCog)
CogLoad("../../WindowCog" WindowCog)
CogLoad("../../PannelCog" PannelCog)
CogLoad("../ExampleDockCog" ExampleDockCog)

CogRequireCogsDomains(${PROJECT_NAME} PrimeMover)
CogRequireCogsDomains(${PROJECT_NAME} WindowCog)
CogRequireCogsDomains(${PROJECT_NAME} PannelCog)
CogRequireCogs(${PROJECT_NAME} ExampleCog)

CogStarterSetEntryPoint(PrimeMover)
CogStarterSetType(Static)
CogStarterAddCogs(${PROJECT_NAME})
CogStarterAddCogs(RaylibCog)
CogStarterAddCogs(PannelCog)
CogStarterAddCogs(ExampleDockCog)
CogStarterCreate()
